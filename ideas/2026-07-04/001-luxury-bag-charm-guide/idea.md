### Data Model (Codable Swift Structs)

```swift
struct Charm: Codable, Identifiable {
    let id: String  // e.g., "hermes-kelly-star"
    let name: String  // e.g., "Kelly Star Charm"
    let brand: String  // e.g., "Hermès"
    let category: String  // e.g., "Animal", "Symbol", "Letter"
    let material: String  // e.g., "Gold", "Silver", "Enamel"
    let priceRange: String  // e.g., "$500-$1000"
    let releaseSeason: String  // e.g., "SS2026", "FW2025"
    let compatibleBags: [String]  // e.g., ["Kelly", "Birkin"]
    let description: String
    let imageNames: [String]
}

struct Bag: Codable, Identifiable {
    let id: String  // e.g., "kelly-25"
    let name: String  // e.g., "Kelly 25"
    let brand: String  // e.g., "Hermès"
    let dimensions: String  // e.g., "25cm x 15cm x 10cm"
    let charmSlots: Int  // Number of charm attachment points
    let compatibleCharmTypes: [String]
}

struct StyleGuide: Codable, Identifiable {
    let id: String  // e.g., "minimalist"
    let title: String  // e.g., "Minimalist Charm Combinations"
    let description: String
    let featuredCharms: [String]  // Charm IDs
    let imageName: String
}
```
